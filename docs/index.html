function processAuditStream(heartbeats, ignoredWarningLogs) { const devices = {}; heartbeats.forEach(hb => { const devName = hb.device || "Primary Device"; if (!devices[devName]) devices[devName] = []; devices[devName].push(hb); }); let noticeHtml = ""; let hasAuditNotice = false; Object.entries(devices).forEach(([devName, beats]) => { // Sort heartbeats chronologically beats.sort((a, b) => new Date(a.timestamp) - new Date(b.timestamp)); const unmonitoredWindows = []; let currentWindowStart = null; beats.forEach((hb, idx) => { const hbTime = new Date(hb.timestamp); if (!hb.incognitoAllowed) { // Start a new unmonitored window if not already inside one if (!currentWindowStart) { currentWindowStart = hbTime; } // If this is the latest heartbeat in log and still false, leave open as active session if (idx === beats.length - 1) { unmonitoredWindows.push({ start: currentWindowStart, end: null }); } } else { // Protection turned ON: Close current unmonitored window if one was active if (currentWindowStart) { unmonitoredWindows.push({ start: currentWindowStart, end: hbTime }); currentWindowStart = null; } } }); if (unmonitoredWindows.length > 0) { hasAuditNotice = true; noticeHtml += `
⚠️ ${devName} — Unmonitored Incognito Window(s):
    `; unmonitoredWindows.forEach(win => { const startStr = formatTime(win.start.toISOString()); if (win.end) { const endStr = formatTime(win.end.toISOString()); const diffMs = win.end - win.start; const diffMin = Math.round(diffMs / 60000); let durationStr = ""; if (diffMin < 1) { durationStr = "< 1 min"; } else if (diffMin < 60) { durationStr = `${diffMin} min`; } else { const hrs = (diffMin / 60).toFixed(1); durationStr = `${hrs} hr`; } noticeHtml += `
  • ${startStr} – ${endStr} (${durationStr})
    • `; } else { noticeHtml += `
  • ${startStr} – Present (Active Unmonitored Session)
    • `; } }); noticeHtml += `
`; } }); if (ignoredWarningLogs.length > 0) { hasAuditNotice = true; noticeHtml += `
🛑 Restricted Sites Visited (${ignoredWarningLogs.length} Ignored Warnings):
    `; ignoredWarningLogs.forEach(log => { const timeStr = formatTime(log.timestamp); const dev = log.device || "Desktop"; const title = log.title ? log.title.replace("[VISITED]", "").trim() : "Untitled"; noticeHtml += `
  • ${timeStr} (${dev}) — "${title}"
    ${log.url}
    • `; }); noticeHtml += `
`; } const noticeContainer = document.getElementById("auditNoticeContainer"); if (hasAuditNotice) { noticeContainer.className = "audit-notice"; noticeContainer.innerHTML = `

⚠️ ATTENTION REQUIRED

${noticeHtml}`; } else { noticeContainer.className = "hidden"; } }
